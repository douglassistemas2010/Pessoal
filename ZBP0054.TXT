  METHOD get_instance.
    IF o_instance IS NOT BOUND.
      o_instance = NEW #( ).
      o_instance->v_grid_visible = abap_false.
      o_instance->v_tree_visible = abap_false.
      IF p_grid EQ abap_true.
        o_instance->v_grid_visible = abap_true.
      ELSEIF p_tree EQ abap_true.
        o_instance->v_tree_visible = abap_true.
      ENDIF.
    ENDIF.
    ro_instance = o_instance.
  ENDMETHOD.

  METHOD run_report.

                        iv_processed = 1
                        iv_total     = 1 ).

    v_is_manager = abap_false.
    SELECT @abap_true
      UP TO 1 ROWS
      INTO @v_is_manager

     WHERE bname  EQ @sy-uname
       AND active EQ @abap_true.

    v_is_consultant = abap_false.

    IF v_is_manager IS INITIAL.

      SELECT @abap_true
        UP TO 1 ROWS
        INTO @v_is_consultant
        FROM /sproagri/tcon02
       WHERE uname  EQ @sy-uname.
      ENDSELECT.
*JAM01{
    ENDIF.
*JAM01}

    " First get involved plants
    DATA(lr_plants) = get_plants( ).
    IF v_is_manager EQ abap_true.
      APPEND LINES OF get_manager_plants( ) TO lr_plants.
    ENDIF.
    SORT lr_plants.
    DELETE ADJACENT DUPLICATES FROM lr_plants.

    IF p_agri IS NOT INITIAL.
      " Get grounds from these plants
      DATA(lt_grounds) = get_grounds( lr_plants ).
      " From ground list, get ground partners, consultants and managers relations
      DATA(lt_relations) = get_ground_relations( lt_grounds ).
      " Build ALV table
      progress_indicator( iv_text      = CONV #( 'Building ALV...'(003) )
                        iv_processed = 1
                        iv_total     = 1 ).
      t_alv = create_alv_table( it_grounds   = lt_grounds
                                it_relations = lt_relations ).
    ELSEIF p_ind IS NOT INITIAL.
      ""Industria - Inicio
      DATA(lt_customers) = get_industry_customers( ).
      ""APPEND LINES OF lt_customers TO lt_grounds.

        DATA(lt_custumers_relations) = get_customer_relations( lt_customers ).
        ""APPEND LINES OF lt_custumers_relations TO lt_relations.
        " Build ALV table
        progress_indicator( iv_text      = CONV #( 'Building ALV...'(003) )
                          iv_processed = 1
                          iv_total     = 1 ).
        t_alv = create_alv_table_industry( it_customers  = lt_customers
                                           it_relations = lt_custumers_relations ).
      ENDIF.
      ""Industria - Fim
    ENDIF.

    FREE lt_grounds.
    FREE lt_relations.

    IF lines( t_alv ) GT 0.
      CALL SCREEN 9000.
    ELSE.
      MESSAGE i398(00) WITH 'No data found.'(001).
    ENDIF.

  ENDMETHOD.

  METHOD get_manager_plants.

    SELECT 'I' AS sign, 'EQ' AS option, relat1~partner2 AS low, relat1~partner2 AS high
      INTO CORRESPONDING FIELDS OF TABLE @rr_plants
      FROM but050 AS relat1
      INNER JOIN /sproagri/tman01 AS tman
      ON relat1~partner1          EQ tman~idman
      INNER JOIN /sproagri/tcbp02 AS relat2
      ON relat1~reltyp            EQ relat2~rtype
      INNER JOIN /sproagri/tcfg05 AS cfg1
      ON relat1~partner2          EQ cfg1~werks
     WHERE tman~idman             EQ @sy-uname
       AND relat2~entit           EQ '09'
       AND relat1~partner2        IN @s_werks
       AND relat1~date_from       LE @sy-datum
       AND relat1~date_to         GE @sy-datum
       AND cfg1~activ             EQ 'X'.

  ENDMETHOD.

  METHOD get_plants.

    SELECT 'I' AS sign, 'EQ' AS option, werks AS low, werks AS high
      INTO CORRESPONDING FIELDS OF TABLE @rr_plants
      FROM t001w
     WHERE werks IN @s_werks.

  METHOD add_node.

    READ TABLE t_tree_keys
      TRANSPORTING NO FIELDS
      WITH TABLE KEY k
      COMPONENTS key = iv_key.

    IF sy-subrc IS INITIAL.
      RETURN.
    ENDIF.

    IF iv_parent_key IS NOT INITIAL.
      READ TABLE t_tree_keys
        ASSIGNING FIELD-SYMBOL(<lf_tree_key>)
        WITH TABLE KEY k
        COMPONENTS key = iv_parent_key.

      IF sy-subrc IS NOT INITIAL.
        RETURN.
      ENDIF.
    ENDIF.

    DATA(lo_node) = io_nodes->add_node( related_node   = COND #( WHEN iv_parent_key EQ ''
                                                                   THEN CONV #( '' )
                                                                   ELSE <lf_tree_key>-node_key )
                                        relationship   = COND #( WHEN iv_parent_key EQ ''
                                                                   THEN if_salv_c_node_relation=>parent
                                                                   ELSE if_salv_c_node_relation=>last_child )
                                        collapsed_icon = CONV #( iv_icon )
                                        expanded_icon  = CONV #( iv_icon )
                                        text           = CONV #( iv_desc ) ).
    lo_node->set_data_row( is_data ).

    INSERT VALUE ty_tree_key( key        = iv_key
                              parent_key = iv_parent_key
                              node_key   = lo_node->get_key( )
                              node_type  = iv_node_type
                              node_obj   = lo_node
                              node_data  = is_data ) INTO TABLE t_tree_keys.

    IF iv_update_hier EQ abap_true.
      update_hier( iv_key      = iv_parent_key
                   iv_leaf_key = iv_key
                   is_data     = is_data ).
    ENDIF.

  ENDMETHOD.

  METHOD update_hier.

    READ TABLE t_tree_totals_ctrl
      TRANSPORTING NO FIELDS
      WITH TABLE KEY key      = iv_key
                     leaf_key = iv_leaf_key.

    IF sy-subrc IS INITIAL.
      RETURN.
    ENDIF.

    READ TABLE t_tree_keys
      ASSIGNING FIELD-SYMBOL(<lf_tree_key>)
      WITH TABLE KEY k
      COMPONENTS key = iv_key.

    IF sy-subrc IS NOT INITIAL.
      RETURN.
    ENDIF.

    <lf_tree_key>-node_data-prod_area  = <lf_tree_key>-node_data-prod_area  + is_data-prod_area.
    IF <lf_tree_key>-node_data-prod_unit IS INITIAL.
      <lf_tree_key>-node_data-prod_unit  = is_data-prod_unit.
    ENDIF.
    <lf_tree_key>-node_data-total_area = <lf_tree_key>-node_data-total_area + is_data-total_area.
    IF <lf_tree_key>-node_data-total_unit IS INITIAL.
      <lf_tree_key>-node_data-total_unit = is_data-total_unit.
    ENDIF.

    INSERT VALUE #( key      = iv_key
                    leaf_key = iv_leaf_key ) INTO TABLE t_tree_totals_ctrl.

    update_hier( iv_key      = <lf_tree_key>-parent_key
                 iv_leaf_key = iv_leaf_key
                 is_data     = is_data ).

  ENDMETHOD.

        binary_tab    = lt_bin_content.
    CLEAR lv_xstr_content.

    CALL FUNCTION 'SCMS_BINARY_TO_STRING'
      EXPORTING
        input_length = lv_len
      IMPORTING
        text_buffer  = lv_str_content
      TABLES
        binary_tab   = lt_bin_content
      EXCEPTIONS
        failed       = 1
        OTHERS       = 2.

    IF sy-subrc IS NOT INITIAL.
      RETURN.
    ENDIF.

    FREE lt_bin_content.

    DATA(lv_json_mapdata) = /ui2/cl_json=>serialize( EXPORTING data = iv_mapdata ).

    REPLACE ALL OCCURRENCES OF '"' IN lv_json_mapdata WITH '\"'.
    REPLACE FIRST OCCURRENCE OF '&&&&' IN lv_str_content WITH lv_json_mapdata.

    CLEAR lv_json_mapdata.

    CALL FUNCTION 'CONVERT_STRING_TO_TABLE'
      EXPORTING
        i_string         = lv_str_content
        i_tabline_length = 1024
      TABLES
        et_table         = lt_data_json.

    CLEAR lv_str_content.

    CALL METHOD o_map->load_data
      EXPORTING
        url                    = 'document.html'
        type                   = 'text'
        subtype                = 'html'
      IMPORTING
        assigned_url           = lv_doc_url
      CHANGING
        data_table             = lt_data_json
      EXCEPTIONS
        dp_invalid_parameter   = 1
        dp_error_general       = 2
        cntl_error             = 3
        html_syntax_notcorrect = 4
        OTHERS                 = 5.
    IF sy-subrc <> 0.
*     MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
*                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
    ENDIF.

    FREE lt_data_json.

    CALL METHOD o_map->show_url
      EXPORTING
        url = lv_doc_url.

  ENDMETHOD.

  METHOD call_bp_display.

    CALL FUNCTION 'FKK_BUPA_DISPLAY'
      EXPORTING
        i_partner        = iv_partner
      EXCEPTIONS
        wrong_parameters = 1
        OTHERS           = 2.

    IF sy-subrc IS NOT INITIAL.
    ENDIF.

  ENDMETHOD.

  METHOD progress_indicator.
    CALL FUNCTION 'PROGRESS_INDICATOR'
      EXPORTING
        i_text               = iv_text
        i_processed          = iv_processed
        i_total              = iv_total
        i_output_immediately = 'X'.
  ENDMETHOD.

  METHOD update_nodes_data.
*TA01{
    DATA(lv_total) = lines( t_tree_keys ).
    DATA(lv_count) = 0.

    LOOP AT t_tree_keys ASSIGNING FIELD-SYMBOL(<lf_tree_key>).

      DATA(lv_processed) = sy-tabix.
      ADD 1 TO lv_count.

      IF lv_count GE 500.
        lv_count = 0.
        progress_indicator( iv_text      = CONV #( 'Preparing Report...'(005) )
                            iv_processed = lv_processed
                            iv_total     = lv_total ).
      ENDIF.
*TA01}
      <lf_tree_key>-node_obj->set_data_row( <lf_tree_key>-node_data ).
    ENDLOOP.
  ENDMETHOD.

  METHOD on_hotspot_click.

    READ TABLE t_alv
      ASSIGNING FIELD-SYMBOL(<lf_alv>)
      INDEX row.

    IF sy-subrc IS NOT INITIAL.
      RETURN.
    ENDIF.

    CASE column.
      WHEN 'CONSULTANT'.
        call_bp_display( CONV #( <lf_alv>-consultant ) ).

      WHEN 'CUSTOMER'.
        call_bp_display( CONV #( <lf_alv>-customer ) ).

      WHEN 'GROUND'.
        call_bp_display( CONV #( <lf_alv>-ground ) ).

      WHEN 'MANAGER'.
        call_bp_display( CONV #( <lf_alv>-manager ) ).

    ENDCASE.

  ENDMETHOD.

ENDCLASS.
